option(ESF_PETSC "Distributed PETSc potential solver" OFF)
option(ESF_OPENMP "Thread independent AMR/molecular kernels (MPI calls stay on main thread)" OFF)
if(ESF_OPENMP)
  if(NOT ESF_PETSC OR NOT ESF_MPI)
    message(FATAL_ERROR "ESF_OPENMP currently requires ESF_MPI and ESF_PETSC")
  endif()
  find_package(OpenMP REQUIRED COMPONENTS C)
  target_link_libraries(esf3d PRIVATE OpenMP::OpenMP_C)
endif()
if(ESF_PETSC)
  if(NOT ESF_MPI)
    message(FATAL_ERROR "ESF_PETSC requires ESF_MPI")
  endif()
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(PETSC REQUIRED IMPORTED_TARGET PETSc)
  # PETSc may export -std=c99 in pkg-config. This project requires C11;
  # retain its include/library options without overriding the selected dialect.
  get_target_property(ESF_PETSC_OPTIONS PkgConfig::PETSC INTERFACE_COMPILE_OPTIONS)
  if(ESF_PETSC_OPTIONS)
    list(FILTER ESF_PETSC_OPTIONS EXCLUDE REGEX "^-std=")
    set_property(TARGET PkgConfig::PETSC PROPERTY INTERFACE_COMPILE_OPTIONS "${ESF_PETSC_OPTIONS}")
  endif()
  include(CheckCSourceCompiles)
  set(CMAKE_REQUIRED_INCLUDES ${PETSC_INCLUDE_DIRS})
  check_c_source_compiles("#include <petscconf.h>
#if defined(PETSC_USE_COMPLEX) || !defined(PETSC_USE_REAL_DOUBLE)
#error ESF requires real double precision PETSc
#endif
int main(void){return 0;}" ESF_PETSC_REAL_DOUBLE)
  unset(CMAKE_REQUIRED_INCLUDES)
  if(NOT ESF_PETSC_REAL_DOUBLE)
    message(FATAL_ERROR "ESF requires a real double precision PETSc build")
  endif()
  target_sources(esf3d PRIVATE src/petsc_potential.c src/amr3_macro.c src/amr3_mesh.c src/amr3_potential.c src/amr3_tags.c src/amr3_structure.c src/amr3_cache.c src/amr3_adapt.c src/amr3_restart.c src/amr3_checkpoint2.c src/amr3_night.c src/amr3_migration.c src/amr3_molecular.c src/amr3.c)
  set(ESF_NIGHT_HASHES "")
  file(GLOB ESF_NUMERICAL_FILES CONFIGURE_DEPENDS src/*.c src/*.h include/*.h)
  list(SORT ESF_NUMERICAL_FILES)
  foreach(name IN LISTS ESF_NUMERICAL_FILES)
    file(SHA256 ${name} hash)
    string(APPEND ESF_NIGHT_HASHES ${hash})
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${name})
  endforeach()
  string(SHA256 ESF_NIGHT_SHA ${ESF_NIGHT_HASHES})
  target_compile_definitions(esf3d PRIVATE ESF_NIGHT_ID="${ESF_NIGHT_SHA}")
  target_link_libraries(esf3d PRIVATE night)
  file(SHA256 ${CMAKE_CURRENT_SOURCE_DIR}/src/geometry.c ESF_GEOMETRY_SHA)
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
    ${CMAKE_CURRENT_SOURCE_DIR}/src/geometry.c ${CMAKE_CURRENT_SOURCE_DIR}/src/igrf.c)
  file(SHA256 ${CMAKE_CURRENT_SOURCE_DIR}/src/igrf.c ESF_IGRF_SHA)
  target_compile_definitions(esf3d PRIVATE ESF_GEOMETRY_ID="${ESF_GEOMETRY_SHA}${ESF_IGRF_SHA}")
  target_compile_definitions(esf3d PRIVATE ESF_PETSC=1)
  target_link_libraries(esf3d PRIVATE PkgConfig::PETSC)
  add_executable(test_petsc tests/test_petsc.c src/transport.c src/petsc_potential.c)
  target_link_libraries(test_petsc PRIVATE tube MPI::MPI_C PkgConfig::PETSC)
  target_compile_definitions(test_petsc PRIVATE ESF_MPI=1 ESF_PETSC=1)
  target_compile_options(test_petsc PRIVATE -Wall -Wextra -Wpedantic)
  add_test(NAME petsc_matrix COMMAND test_petsc)
endif()
